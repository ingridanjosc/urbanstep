import { createClient } from "npm:@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; function json(data: unknown, status = 200) { return new Response(JSON.stringify(data), { status, headers: { ...corsHeaders, "Content-Type": "application/json", }, }); } const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!; const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!; const FRENET_TOKEN = Deno.env.get("FRENET_TOKEN"); const db = createClient( SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY ); Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders, }); } if (req.method !== "POST") { return json( { error: "Método não permitido.", }, 405 ); } try { if (!FRENET_TOKEN) { return json( { error: "Token da Frenet não configurado.", }, 500 ); } const body = await req.json(); const produtoId = Number(body.produto_id); const cepDestino = String(body.cep || "") .replace(/\D/g, ""); const quantidade = Math.max( 1, Number(body.quantidade || 1) ); if (!produtoId) { return json( { error: "Produto não informado.", }, 400 ); } if (cepDestino.length !== 8) { return json( { error: "Informe um CEP válido.", }, 400 ); } /* BUSCA PRODUTO + FORNECEDOR */ const { data: produto, error: produtoError, } = await db .from("produtos") .select(` id, nome, marca, preco, preco_promocional, ativo, fornecedor_id, fornecedores ( id, nome, cep ) `) .eq("id", produtoId) .eq("ativo", true) .single(); if ( produtoError || !produto ) { console.error( produtoError ); return json( { error: "Produto não encontrado.", }, 404 ); } const fornecedor: any = produto.fornecedores; if (!fornecedor) { return json( { error: "Este produto ainda não possui fornecedor cadastrado.", }, 400 ); } const cepOrigem = String( fornecedor.cep || "" ) .replace(/\D/g, ""); if (cepOrigem.length !== 8) { return json( { error: "O CEP do fornecedor é inválido.", }, 400 ); } const precoProduto = Number( produto.preco_promocional || produto.preco ); if ( !Number.isFinite( precoProduto ) || precoProduto <= 0 ) { return json( { error: "Preço do produto inválido.", }, 400 ); } /* COTAÇÃO FRENET PADRÃO DOS TÊNIS: 10 × 20 × 30 cm 800 g */ const frenetPayload = { SellerCEP: cepOrigem, RecipientCEP: cepDestino, ShipmentInvoiceValue: precoProduto * quantidade, ShippingItemArray: [ { Height: 10, Width: 20, Length: 30, Weight: 0.8, Quantity: quantidade, SKU: String(produto.id), }, ], }; const frenetResponse = await fetch( "https://api.frenet.com.br/shipping/quote", { method: "POST", headers: { "Content-Type": "application/json", token: FRENET_TOKEN, }, body: JSON.stringify( frenetPayload ), } ); const frenetData = await frenetResponse.json(); if (!frenetResponse.ok) { console.error( "Erro Frenet:", frenetData ); return json( { error: frenetData?.Message || frenetData?.message || "Não foi possível calcular o frete.", }, frenetResponse.status ); } const servicos = frenetData .ShippingSevicesArray || frenetData .ShippingServicesArray || []; /* LIMPA A RESPOSTA PARA O SITE */ const opcoes = servicos .filter( (item: any) => { const preco = Number( item.ShippingPrice || 0 ); return ( !item.Error && Number.isFinite(preco) && preco >= 0 ); } ) .map( (item: any) => ({ codigo: String( item.ServiceCode || item.ShippingServiceCode || "" ), servico: item.ServiceDescription || item.ShippingServiceName || "Entrega", transportadora: item.Carrier || item.CarrierName || "", /* Esse é o custo REAL que você terá na Frenet. */ preco: Number( item.ShippingPrice || 0 ), /* Para o consumidor: frete é GRÁTIS. */ preco_cliente: 0, gratis: true, prazo: Number( item.DeliveryTime || 0 ), }) ); if (!opcoes.length) { return json( { error: "Nenhuma opção de entrega disponível para este CEP.", }, 404 ); } return json({ success: true, frete_gratis: true, fornecedor: fornecedor.nome, cep_origem: cepOrigem, cep_destino: cepDestino, opcoes, }); } catch (error) { console.error(error); return json( { error: error instanceof Error ? error.message : "Erro interno ao calcular frete.", }, 500 ); } });
